/*****声明仓库管理结构体*****/
typedef struct{
	char num[10]; //货物编号
	char name[20]; //货物名称
	char tele[12]; //货物数量
	char place[20]; //存放地点
	char peoplename[50]; //保管员姓名
}Address;


/*****声明单链表结构*****/
typedef struct Node{
	Address data;
	struct Node *next;
}LNode;

/*****声明单链表指针*****/
typedef LNode *LinkList;

/*****定义单链表S*****/
LinkList S;

int Add_Num; //单链表的数据个数，即联系人的个数。

/*****显示货物信息*****/
void Show(LinkList L){
	LinkList p=L;
system("cls");
printf("管理系统共有%d个货物\n\n",Add_Num);
printf("货物编号\t货物名称\t货物数量\t存放地点\t保管员姓名\n");
while(p->next){
p=p->next;
printf("%-8s%-8s%-8s%-8s%-15s\n",
p->data.num,p->data.name,p->data.tele,p->data.place,p->data.peoplename);
}
printf("\n");
printf("\n按任意键返回菜单!");
getch();
}

/*****新建仓库信息*****/
//尾插法创建单链表结构存储货物信息
void New_One(){
	int i;
	LinkList p,r;//分配头结点存储空间
	S=(LinkList)malloc(sizeof(LNode));
	if(S==NULL){//判断是否分配成功
		printf("内存空间分配失败!");
		exit(0);
	}
	S->next=NULL;
	r=S;//r是尾指针，指内最后一个结点
	printf("\n请输入新建货物的数目\n");
	scanf("%d",&Add_Num);//输入新建货物数
	for(i=0;i<Add_Num;i++){
		p=(LinkList)malloc(sizeof(LNode));//分配新结点存储空间
		if(p==NULL){
		printf("内存空间分配失败!");
		exit(0);
}
		system("cls");//清屏
		printf("请输入第%d个货物信息:\n",i+1);
		printf("请输入货物编号:");
		scanf("%s",p->data.num);
		printf("\n请输入货物名称:");
		scanf("%s",p->data.name);
		printf("\n请输入货物数量:");
		scanf("%s",p->data.tele);
		printf("\n请输入存放地点:");
		scanf("%s",p->data.place);
		printf("\n请输入保管员姓名:");
		scanf("%s",p->data.peoplename);
		p->next=NULL;//当前新结点是最后一个结点
		r->next=p;//将新结点插入到表尾
		r=p;//尾指针指向新结点
}
	printf("\n输入成功!");
	Show(S);//显示链表所有信息
}




/*****添加联系人*****/

	int i=0;

	while(p&&i<Add_Num){
		p=p->next;

}
	r=(LinkList)malloc(sizeof(LNode));

	printf("\n请输入货物数量:");
	scanf("%s",r->data.tele);
	printf("\n请输入存放地点:");
	scanf("%s",r->data.place);
	printf("\n请输入保管员姓名:");
	scanf("%s",r->data.peoplename);
	p->next=r;
	r->next=NULL;
	Add_Num++;
	printf("\n输入成功!");
	Show(L);
}

/*****删除货物信息*****/
void Delete(LinkList L){
	LinkList p,s;//判断是否找到了被删除数据
	int flag_tail=1;
	char number[10];
	system("cls");printf("请输入要删除的编号:\n");
	scanf("%s",number);
	p=L;
	while(p->next){
		if(strcmp(p->next->data.num,number)==0){
			s=p->next;p->next=s->next;free(s);
			Add_Num--;
			flag_tail=0;
			break;
		}
	p=p->next;
	}
		if(flag_tail){
			printf("\n该数据不存在，无法删除!\n");printf("\n");
			printf("\n按任意键返回菜单!");
			getch();
	}
		else{
			printf("\n删除成功!\n");
			printf("\n");
			printf("\n按任意键返回菜单!");
			getch();
	}
}

/*****排序函数*****/
//按货物名称从小到大排序
void Sort(LinkList L){
	int i,j;
	LinkList p;
	Address tmp;
	for(i=0;i<Add_Num-1;i++){
		p=L->next;
		for(j=0;j<Add_Num-i-1;j++){
			if(strcmp(p->data.name,p->next->data.name)>0){
				tmp=p->data;
				p->data=p->next->data;
				p->next->data=tmp;
			}
			p=p->next;
		}
	}
	Show(L);
}







/*****打开文件*****/
void Open(){
	char filename[30];
	FILE *fp;
	LinkList p,r;
	int i;
	S=(LinkList)malloc(sizeof(LNode));
	if(S==NULL){
		printf("内存空间分配失败!");
		exit(0);
	}
	S->next=NULL;
	r=S;
	printf("请输入要打开的文件名:\n");
	scanf("%s",filename);
	if((fp=fopen(filename,"rb+"))==NULL){
		printf("文件不能打开，按任意键退出\n");
		exit(0);
	}
	fread(&Add_Num,sizeof(int),1,fp);
	for(i=0;i<Add_Num;i++){
	p=(LinkList)malloc(sizeof(LNode));
	if(p==NULL){
		printf("内存空间分配失败!");
		exit(0);
	}
	fread(&(p->data),sizeof(Address),1,fp);
	p->next=NULL;
	r->next=p;
	r=p;
	}
	fclose(fp);
	printf("\n文件信息导入成功!\n");
	printf("\n按任意键返回菜单!");
	getch();
}

/*****保存文件*****/
void Save(){
	LinkList r;
	char filename[30];
	FILE *fp;//文件指针
	printf("请输入要保存的文件名\n");
	scanf("%s",filename);
	if((fp=fopen(filename,"wb"))==NULL){ //用”写”的方式打开文件
		printf("文件创建失败，按任意键退出\n");
		exit(0);
}
	fwrite(&Add_Num,sizeof(int),1,fp);
	r=S->next;
	while(r!=NULL){
		fwrite(&(r->data),sizeof(Address),1,fp);
		r=r->next;
	}
	fclose(fp);
	printf("\n文件保存成功!\n");
	printf("\n按任意键返回菜单!");
	getch();
}
